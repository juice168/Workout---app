  const counter = document.createElement('div');
  counter.className = 'exercise-counter';
  counter.textContent = `Exercise ${currentExIdx + 1} of ${day.exercises.length}`;
  area.appendChild(counter);

  // Name
  const name = document.createElement('div');
  name.className = 'exercise-name';
  name.textContent = ex.name;
  area.appendChild(name);

  // Reps info
  const repsInfo = document.createElement('div');
  repsInfo.className = 'reps-info';
  repsInfo.textContent = `${ex.sets} sets · ${ex.reps} reps · 75 sec rest`;
  area.appendChild(repsInfo);

  // Sets
  const setsRow = document.createElement('div');
  setsRow.className = 'sets-row';

  for (let s = 0; s < ex.sets; s++) {
    const chip = document.createElement('div');
    chip.className = 'set-chip' + (setsCompleted[key][s] ? ' done' : '');
    chip.innerHTML = `<div class="set-dot"></div>Set ${s + 1}`;
    chip.addEventListener('click', () => toggleSet(key, s, ex.sets, ex.name));
    setsRow.appendChild(chip);
  }
  area.appendChild(setsRow);

  // Cardio — show on last exercise
  if (currentExIdx === day.exercises.length - 1) {
    const box = document.createElement('div');
    box.className = 'cardio-box';
    box.innerHTML = `<div class="cardio-label">📍 Post-Workout Cardio</div><div class="cardio-text">${day.cardio}</div>`;
    area.appendChild(box);
  }

  // Prev/Next
  document.getElementById('btnPrev').disabled = currentExIdx === 0;
  const isLast = currentExIdx === day.exercises.length - 1;
  document.getElementById('btnNext').textContent = isLast ? 'FINISH 🏁' : 'NEXT →';
}

// ── TOGGLE SET ──
function toggleSet(exKey, setIdx, totalSets, exName) {
  const wasCompleted = setsCompleted[exKey][setIdx];
  setsCompleted[exKey][setIdx] = !wasCompleted;

  if (!wasCompleted) {
    totalSetsCount++;
    // Show rest timer if not last set
    const allDone = setsCompleted[exKey].every(Boolean);
    const nextSetIdx = setIdx + 1;
    if (nextSetIdx < totalSets) {
      showRest(`Set ${nextSetIdx + 1} of ${totalSets} · ${exName}`);
    } else if (!allDone) {
      showRest(`Next: ${getNextExName()}`);
    }
  }

  renderExercise();
}

function getNextExName() {
  const day = DAYS[currentDay];
  if (currentExIdx + 1 < day.exercises.length) {
    return day.exercises[currentExIdx + 1].name;
  }
  return 'Cardio 💪';
}

// ── REST TIMER ──
function showRest(nextLabel) {
  document.getElementById('restNext').innerHTML = `Up next: <strong>${nextLabel}</strong>`;
  restRemaining = REST_DURATION;
  updateRingAndTime();
  showScreen('rest');

  if (restTimer) clearInterval(restTimer);
  restTimer = setInterval(() => {
    restRemaining--;
    updateRingAndTime();
    if (restRemaining <= 0) {
      clearInterval(restTimer);
      // Vibrate if available
      if (navigator.vibrate) navigator.vibrate([200, 100, 200]);
      hideRest();
    }
  }, 1000);
}

function updateRingAndTime() {
  document.getElementById('restSeconds').textContent = restRemaining;
  const pct = restRemaining / REST_DURATION;
  const offset = CIRC * (1 - pct);
  document.getElementById('ringFill').style.strokeDashoffset = offset;
  const hue = pct > 0.5 ? 200 : pct > 0.25 ? 45 : 10;
  document.getElementById('ringFill').style.stroke = `hsl(${hue}, 90%, 65%)`;
}

function skipRest() {
  if (restTimer) clearInterval(restTimer);
  hideRest();
}

function hideRest() {
  document.getElementById('rest').classList.remove('active');
  document.getElementById('rest').style.display = 'none';
  // Restore workout screen
  document.getElementById('workout').style.display = 'flex';
  document.getElementById('workout').classList.add('active');
}

// ── NAV ──
function prevExercise() {
  if (currentExIdx > 0) { currentExIdx--; renderExercise(); }
}

function nextExercise() {
  const day = DAYS[currentDay];
  if (currentExIdx < day.exercises.length - 1) {
    currentExIdx++;
    renderExercise();
  } else {
    finishWorkout();
  }
}

function finishWorkout() {
  const day = DAYS[currentDay];
  const elapsed = Math.round((Date.now() - startTime) / 60000);
  const allSets = day.exercises.reduce((sum, ex) => sum + ex.sets, 0);

  document.getElementById('completeDay').textContent = day.label + ' · ' + day.title + ' Complete';
  document.getElementById('statSets').textContent = totalSetsCount;
  document.getElementById('statTime').textContent = elapsed;
  document.getElementById('statEx').textContent = day.exercises.length;

  document.getElementById('progressFill').style.width = '100%';
  showScreen('complete');
}

function goHome() {
  showScreen('home');
}

  document.getElementById(id).style.display = 'flex';
}
